# golden_age scenario — boot into gameplay on the "Golden Age" scenario at Deity
# difficulty as the Russians (deterministic + tutorial-free). Chosen over Beta
# Centauri because it does not open with the Modern-Era popup.
#
# Robust navigation (D-pad step-counts are unreliable — the pad drops/auto-repeats
# intermittently): list/carousel selection is OCR-VERIFIED via `find_text` with a
# --crop over the description panel, which shows the SELECTED item's name (so a
# match means selected, not merely visible). Plain menus (main menu, Single Player
# submenu, difficulty) use small fixed counts at the 0.2s D-pad hold, each
# followed by a wait_text that verifies we landed on the right screen.

# --- boot to main menu ---
wait  Hardware scaler|Shader .* translated successfully  90
wait_text  Press START  90
input START
sleep 3
input B                       # decline sign-in -> main menu
sleep 3
shot 01_main_menu

# --- main menu -> Single Player (1 Down, A), verify submenu ---
input DPAD_DOWN
sleep 1
input A
wait_text  New Game|Play Scenario|Game of the Week  25
shot 02_single_player

# --- Single Player -> Play Scenario (find it selected, then A) ---
# The SP submenu has no description panel; scroll down and OCR the whole screen
# for the "Choose Scenario" that appears after selecting Play Scenario. Press A
# after each down until the scenario list loads.
input DPAD_DOWN
sleep 0.6
input DPAD_DOWN
sleep 0.6
input DPAD_DOWN
sleep 0.6
input A
wait_text  Choose Scenario  30
sleep 3                       # let the list finish animating in
shot 03_scenario_list

# --- select Golden Age (OCR-verified via the description-panel header) ---
find_text  DPAD_DOWN  14  crop=0.54,0.21,0.82,0.33  Golden Age
sleep 0.6
shot 04_golden_selected
input A
wait_text  DIFFICULTY|Chieftain  25
shot 05_difficulty

# --- Deity difficulty (4 Down from Chieftain, A) ---
input DPAD_DOWN
sleep 0.5
input DPAD_DOWN
sleep 0.5
input DPAD_DOWN
sleep 0.5
input DPAD_DOWN
sleep 0.6
input A
wait_text  Civilization|Accept  40
sleep 2
shot 06_civ_select

# --- select Russians (OCR-verified: Catherine shows in the civ description) ---
find_text  DPAD_RIGHT  18  crop=0.40,0.50,0.63,0.63  Catherine|Russian
sleep 0.6
shot 07_russians

# --- accept Russians (Accept can need a second press), load the game ---
input A
sleep 2
input A
wait_text  Loading|Galley|Ranger|Golden  40
shot 08_loading
sleep 18
input B                       # dismiss any start-of-game popup
sleep 3
shot 09_gameplay

expect_no_crash
