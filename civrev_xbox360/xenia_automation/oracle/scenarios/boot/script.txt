# boot scenario — take the game from launch to the MAIN MENU.
#
# The intro movies are skipped at the container level (empty bind-mount, see
# container.sh) and a profile is seeded, so boot is fast and DETERMINISTIC:
# render comes up, a brief static legal screen, then the "Press START" title.
# Fixed timing is reliable here (wait_stable is unusable — the title background
# is a live 3D scene that never fully settles).
#
# Flow: wait for render -> title -> START -> decline sign-in (B) -> main menu.
# Input goes through the virtual gamepad (virtpad.py / input.sh).

wait  Hardware scaler|Shader .* translated successfully  90
wait_text  Press START  90   # OCR-detect the title before advancing (robust)
shot 01_title
input START            # -> "Would you like to sign in?"
sleep 3
input B                # A=Yes / B=No -> decline -> main menu
sleep 4
shot 02_main_menu

expect_no_crash
